\begin{frame}{Exponential Clocks Replace Rigid Freezing}
  \begin{columns}[onlytextwidth,c]
    \begin{column}{0.43\textwidth}
      An edge $e=\{i,j\}$ is \textbf{satisfied} when
      \[
        \begin{array}{ll}
          \sigma_i=\sigma_j, & e\in F,\\[2pt]
          \sigma_i\neq\sigma_j, & e\in\widetilde F.
        \end{array}
      \]

      For each satisfied edge, independently,
      \[
        \boxed{\ \xi_e\sim\operatorname{Exp}(|W_e|)\ }.
      \]
      At level $t$, keep the edges such that $\xi_e\leq t$;
      unsatisfied edges are never kept.

      \begin{block}{Classical \textsc{Swendsen}--\textsc{Wang}}
        The usual freezing rule is exactly the cut $t=1$:
        \[
          \mathbb P(\xi_e\leq1)=1-e^{-|W_e|}.
        \]
      \end{block}
    \end{column}

    \boxed{\ \Lambda_u(\sigma)

\begin{frame}{The Exact \textsc{Edwards}--\textsc{Sokal} Lift}
  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Clock--Kruskal law}
        On the support of $D$,
        \[
          \begin{aligned}
          \mathbb P(\mathrm dD\mid\sigma)
          &=e^{-\Lambda_\infty(\sigma)}\\[-2pt]
          &\quad\times\prod_{u\in\operatorname{Int}(D)}
            \Lambda_u(\sigma)e^{-\beta_u\Lambda_u(\sigma)}\,\mathrm dD.
          \end{aligned}
        \]
        Here $\Lambda_\infty(\sigma)$ is the total satisfied weight between
        distinct components at the cut $t=1$.
      \end{block}

      \begin{block}{Joint law}
        \[
          \mu(\mathrm d\sigma,\mathrm dD)
          =\frac{1}{Z}\mu_0(\mathrm d\sigma)e^{-U(\sigma)}
            \mathbb P(\mathrm dD\mid\sigma).
        \]
        Its $\sigma$-marginal is the target Gibbs law.
      \end{block}
    \end{column}

    \begin{column}{0.47\textwidth}
      Since every edge belongs to one merge, or joins two final components,
      \[
        e^{-U(\sigma)}
        \ \propto\
        \exp\!\left\{
          \sum_u\Lambda_u(\sigma)+\Lambda_\infty(\sigma)
        \right\}.
      \]
      The $\Lambda_\infty$ terms cancel at the cut $t=1$. Therefore
      \begin{block}{Conditional law used for the flips}
        {\small
        \[
          \boxed{
          \begin{aligned}
          \mu(\sigma\mid D)
          &\ \propto\ \mu_0(\sigma)\\[-2pt]
          &\quad\times\prod_{u}
          \Lambda_u(\sigma)e^{(1-\beta_u)\Lambda_u(\sigma)}.
          \end{aligned}}
        \]
        }
      \end{block}
      Thus $\mu_0$, all the $W_e$, and the merge clocks
      $\beta_u=\xi_{e_u}$ are retained exactly.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{One Split Has Four Possible Outcomes}
  At $u:C=C_1\mathbin{\dot\cup}C_2$, write $\sigma^{ab}$ for the configuration
  obtained by flipping $C_1$ when $a=1$ and $C_2$ when $b=1$.

  \vspace{0.05em}
  \begin{center}
    \renewcommand{\arraystretch}{1.18}
    \begin{tabular}{c c c c}
      \toprule
      event $(a,b)$ & $C_1$ & $C_2$ & $\Lambda_u(\sigma^{ab})$ \\
      \midrule
      $(0,0)$ & keep & keep & $\Lambda_u(\sigma)$ \\
      $(1,0)$ & flip & keep & $\displaystyle\sum_{e\in E_u}|W_e|-\Lambda_u(\sigma)$ \\
      $(0,1)$ & keep & flip & $\displaystyle\sum_{e\in E_u}|W_e|-\Lambda_u(\sigma)$ \\
      $(1,1)$ & flip & flip & $\Lambda_u(\sigma)$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \textbf{Contribution of the current merge:}
  \[
    \boxed{
    \Lambda_u(\sigma^{ab})
    \exp\!\left\{(1-\xi_{e_u})\Lambda_u(\sigma^{ab})\right\}.}
  \]

  \begin{alertblock}{Local symmetry, four global weights}
    Locally, $a=b$ and $a\neq b$ are the two cases; $\mu_0$ and the ancestors
    distinguish the four weights.
  \end{alertblock}
\end{frame}

\begin{frame}{The Four Probabilities Form One Exact Heat Bath}
  The exact rule at $u$ is
  \[
    \boxed{
      p_u(a,b\mid\sigma,D)
      =\frac{\mu(\sigma^{ab}\mid D)}
             {\displaystyle\sum_{c,d\in\{0,1\}}
              \mu(\sigma^{cd}\mid D)},
      \qquad (a,b)\in\{0,1\}^2.}
  \]

  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.60\textwidth}
      After cancelling the factors common to the four candidates,
      \[
        \mu(\sigma^{ab}\mid D)
        \ \propto\
        \mu_0(\sigma^{ab})
        \prod_{v\succeq u}
        \left[
          \Lambda_v(\sigma^{ab})
          e^{(1-\beta_v)\Lambda_v(\sigma^{ab})}
        \right],
      \]
      where $v\succeq u$ means $v=u$ or $v$ is an ancestor of $u$, and
      $\beta_v=\xi_{e_v}$.
    \end{column}

    \begin{column}{0.36\textwidth}
      \begin{block}{At each visited node}
        \begin{enumerate}
          \item form the four $\sigma^{ab}$;
          \item evaluate $\mu_0(\sigma^{ab})$ and recompute
            $\Lambda_v(\sigma^{ab})$ for $v=u$ and its ancestors;
          \item normalise the four weights;
          \item draw one event and update $\sigma$.
        \end{enumerate}
      \end{block}
    \end{column}
  \end{columns}

  Descendant merges and disjoint branches are unchanged, so their factors cancel.
  Each node uses one joint four-state heat bath; node updates are sequential.
\end{frame}

\begin{frame}{Detailed Balance Holds at Every Visited Node}
  Fix $D$ and a node $u$.  From any of its four configurations, the candidate
  set is the same:
  \[
    \bigl\{\sigma^{00},\sigma^{10},\sigma^{01},\sigma^{11}\bigr\}.
  \]

  For two candidates $x$ and $y$, the heat-bath kernel $K_u^D$ satisfies
  \begin{block}{Detailed balance for the four-state update}
    \[
      \mu(x\mid D)K_u^D(x,y)
      =\frac{\mu(x\mid D)\mu(y\mid D)}
             {\displaystyle\sum_{c,d\in\{0,1\}}\mu(\sigma^{cd}\mid D)}
      =\mu(y\mid D)K_u^D(y,x).
    \]
  \end{block}

  Hence each node update is reversible and preserves $\mu(\,\cdot\mid D)$.
  Zero conditional weight simply gives zero probability to the corresponding event.

  \begin{alertblock}{One useful distinction}
    Every local heat bath satisfies detailed balance.  A deterministic sweep of
    several nodes is stationary, but need not itself be reversible.
  \end{alertblock}
\end{frame}

\begin{frame}{A Full Sweep---or Any Fixed Prefix---Preserves Gibbs}
  Fix a traversal $u_1,\ldots,u_M$ of the full sampled dendrogram $D$, and stop
  after the first $m$ nodes. Since every node heat bath preserves the same
  conditional law,
  \[
    \boxed{
      \mu(\,\cdot\mid D)K_{u_1}^D\cdots K_{u_m}^D
      =\mu(\,\cdot\mid D),
      \qquad m=0,1,\ldots,M.}
  \]

  \begin{center}
    \Large
    $\displaystyle
      \sigma\sim\mu
      \ \xrightarrow{\ D\sim\mathbb P(\,\cdot\mid\sigma)\ }\
      (\sigma,D)
      \ \xrightarrow{\ m\ \text{node heat baths}\ }\
      (\sigma',D)
      \ \xrightarrow{\ \text{discard }D\ }\
      \sigma'\sim\mu.$
  \end{center}

  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Allowed stopping rules}
        Keep the full $D$; traverse the whole forest, or stop at any depth fixed
        in advance, chosen from $D$, or chosen by external randomness.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{What is not automatic}
        A stopping rule depending on the spins produced during the sweep
        requires a separate invariance argument.
      \end{alertblock}
    \end{column}
  \end{columns}

  \begin{center}
    \textbf{Stationarity, not perfect sampling:}
    one sweep need not produce a Gibbs sample from an arbitrary initial state.
  \end{center}
\end{frame}
