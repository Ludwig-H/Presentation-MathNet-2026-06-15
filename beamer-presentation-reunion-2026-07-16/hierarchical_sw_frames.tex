% Hierarchical Swendsen--Wang dynamics.

\begin{frame}{Target Gibbs Law and Exponential Clocks}
  Let $E=F\mathbin{\dot\cup}\widetilde F$.  An edge $e=\{i,j\}$ is
  \textbf{satisfied} when
  \[
    \sigma_i=\sigma_j\quad(e\in F),
    \qquad
    \sigma_i\neq\sigma_j\quad(e\in\widetilde F).
  \]
  Thus, up to an additive constant,
  \[
    U(\sigma)=\sum_{e\in E}|W_e|\ind{e\text{ is not satisfied}},
    \qquad
    \boxed{\ \mu(\sigma)\propto\mu_0(\sigma)e^{-U(\sigma)}.\ }
  \]

  \begin{columns}[onlytextwidth,c]
    \begin{column}{0.55\textwidth}
      For each satisfied edge, independently,
      \[
        \boxed{\ \xi_e\sim\operatorname{Exp}(|W_e|)\ },
      \]
      while $\xi_e=+\infty$ for an unsatisfied edge.  At level $t$, retain
      precisely the links with $\xi_e\leq t$.

      \begin{block}{Classical \textsc{Swendsen}--\textsc{Wang} is the cut $t=1$}
        \[
          \mathbb P(\xi_e\leq1\mid e\text{ satisfied})
          =1-e^{-|W_e|}.
        \]
      \end{block}
    \end{column}

    \begin{column}{0.40\textwidth}

    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Auxiliary Variable Is the Component Dendrogram up to $1$}
  Define the graph and its component partition at every level by
  \[
    G_t=\bigl(V,\{e\in E:\xi_e\leq t\}\bigr),
    \qquad
    \Pi_t=\operatorname{cc}(G_t),
    \qquad
    \boxed{\ D_{\leq1}=(\Pi_t)_{0\leq t\leq1}.\ }
  \]

  \begin{columns}[onlytextwidth,c]
    \begin{column}{0.48\textwidth}
      \begin{block}{Finite description of $D_{\leq1}$}
        \begin{itemize}
          \item the final components (the roots) $\Pi_1$;
          \item for each merger $u:C=C_1\mathbin{\dot\cup}C_2$, the two
            children $C_1,C_2$ and its height $\beta_u\leq1$.
        \end{itemize}
      \end{block}

      Children are unordered.  The support is
      \[
        0<\beta_v<\beta_u\leq1
        \quad\text{whenever $v$ is below $u$.}
      \]
      The reference measure is counting measure on these forest topologies
      times Lebesgue measure on their admissible heights.
    \end{column}

\begin{frame}{Exact Density of the Censored Dendrogram}
  Fix a candidate forest topology.  Assign each graph edge to
  \begin{itemize}
    \item the unique split $E_u$ given by the first common cluster of its endpoints; or
    \item $E_\infty$, if its endpoints lie in distinct roots of $\Pi_1$.
  \end{itemize}
  Hence
  \[
    E=\left(\mathop{\dot\bigcup}_{u\in\operatorname{Int}(D_{\leq1})}E_u\right)
      \mathbin{\dot\cup}E_\infty,
    \qquad
    \Lambda_\infty(\sigma)
      =\sum_{e\in E_\infty}|W_e|\ind{e\text{ satisfied}}.
  \]

  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Internal node $u$}
        The first clock in $E_u$ has density
        \[
          \Lambda_u(\sigma)e^{-\beta_u\Lambda_u(\sigma)}\,d\beta_u.
        \]
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Between final roots}
        No satisfied clock in $E_\infty$ rings before $1$:
        \[
          \mathbb P(\min_{e\in E_\infty}\xi_e>1\mid\sigma)
          =e^{-\Lambda_\infty(\sigma)}.
        \]
      \end{block}
    \end{column}
  \end{columns}

  The clock families in this disjoint partition are independent.  Therefore
  \[
    \boxed{
    \mathbb P(dD_{\leq1}\mid\sigma)
      =\ind{D_{\leq1}\text{ admissible}}
       e^{-\Lambda_\infty(\sigma)}
       \prod_{u\in\operatorname{Int}(D_{\leq1})}
       \Lambda_u(\sigma)e^{-\beta_u\Lambda_u(\sigma)}\,dD_{\leq1}.}
  \]
  A topology with $\Lambda_u(\sigma)=0$ has zero density, as it should.
\end{frame}

\begin{frame}{The \textsc{Edwards}--\textsc{Sokal} Lift and Its Conditional Law}
  Define the joint law on spin--dendrogram pairs by
  \[
    \boxed{
    \nu(d\sigma,dD)
      =\frac{1}{Z}\mu_0(d\sigma)e^{-U(\sigma)}
        \mathbb P(dD_{\leq1}\mid\sigma).}
  \]
  Since $\mathbb P(\,\cdot\mid\sigma)$ is a probability law, the
  $\sigma$-marginal of $\nu$ is exactly the target Gibbs law $\mu$.

  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.47\textwidth}
      Every edge is in one $E_u$ or in $E_\infty$, so
      \[
        \sum_{e\in E}|W_e|\ind{e\text{ satisfied}}
        =\sum_u\Lambda_u(\sigma)+\Lambda_\infty(\sigma).
      \]
      Moreover
      \[
        e^{-U(\sigma)}
        =e^{-\sum_{e\in E}|W_e|}
         e^{\sum_u\Lambda_u(\sigma)+\Lambda_\infty(\sigma)}.
      \]
    \end{column}

    \begin{column}{0.49\textwidth}
      The $\Lambda_\infty$ terms cancel exactly at the cut $1$.  Thus, on the
      support of $\nu$,
      \begin{block}{Conditional law used by every flip}
        \[
          \boxed{
          \nu(\sigma\mid D)
          \propto
          \mu_0(\sigma)
          \prod_{u}
          \Lambda_u(\sigma)
          e^{(1-\beta_u)\Lambda_u(\sigma)}.}
        \]
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{At a Split There Are Four Joint Events}
  For $u:C=C_1\mathbin{\dot\cup}C_2$, let $\sigma^{ab}$ flip $C_1$ when
  $a=1$ and $C_2$ when $b=1$.  In the binary signed model,
  \[
    \begin{array}{c|c|c|c}
      (a,b) & C_1 & C_2 & \Lambda_u(\sigma^{ab}) \\
      \hline
      (0,0) & \text{keep} & \text{keep} & \Lambda_u(\sigma) \\
      (1,0) & \text{flip} & \text{keep} & \displaystyle\sum_{e\in E_u}|W_e|-\Lambda_u(\sigma) \\
      (0,1) & \text{keep} & \text{flip} & \displaystyle\sum_{e\in E_u}|W_e|-\Lambda_u(\sigma) \\
      (1,1) & \text{flip} & \text{flip} & \Lambda_u(\sigma)
    \end{array}
  \]

  \vspace{-0.3em}
  \begin{alertblock}{One four-state draw, not two independent coins}
    The current split only distinguishes equal versus opposite flip parity.
    The prior $\mu_0$ and the ancestor cuts generally distinguish all four events.
  \end{alertblock}

  \vspace{-0.2em}
  If $u$ is a root and $\mu_0$ is uniform, set
  $A_u=\sum_{e\in E_u}|W_e|$.  Then the fully explicit local probabilities are
  \[
    p_{00}=p_{11}
      =\frac{\Lambda_u e^{(1-\beta_u)\Lambda_u}}
      {2\!\left[\Lambda_u e^{(1-\beta_u)\Lambda_u}
      +(A_u-\Lambda_u)e^{(1-\beta_u)(A_u-\Lambda_u)}\right]},
  \]
  \[
    p_{10}=p_{01}
      =\frac{(A_u-\Lambda_u)e^{(1-\beta_u)(A_u-\Lambda_u)}}
      {2\!\left[\Lambda_u e^{(1-\beta_u)\Lambda_u}
      +(A_u-\Lambda_u)e^{(1-\beta_u)(A_u-\Lambda_u)}\right]}.
  \]
\end{frame}

\begin{frame}{The Exact Four Probabilities at an Arbitrary Level}
  For the general node $u$, factors below $u$ and on disjoint branches are
  unchanged.  The exact heat bath is therefore
  \[
    \boxed{
    p_u(a,b\mid\sigma,D)
    =\frac{
      \mu_0(\sigma^{ab})
      \displaystyle\prod_{v\succeq u}
      \left[\Lambda_v(\sigma^{ab})
      e^{(1-\beta_v)\Lambda_v(\sigma^{ab})}\right]}
    {\displaystyle\sum_{c,d\in\{0,1\}}
      \mu_0(\sigma^{cd})
      \prod_{v\succeq u}
      \left[\Lambda_v(\sigma^{cd})
      e^{(1-\beta_v)\Lambda_v(\sigma^{cd})}\right]}.}
  \]
  Here $v\succeq u$ means $v=u$ or $v$ is an ancestor of $u$, and
  \[
    \Lambda_v(\sigma^{ab})
      =\sum_{e\in E_v}|W_e|\ind{e\text{ satisfied by }\sigma^{ab}}.
  \]

  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Where the clock minimum enters}
        The sampled height $\beta_v$ is the minimum crossing clock used to
        construct $D$.  It is then held fixed and appears explicitly in every
        factor $e^{(1-\beta_v)\Lambda_v}$.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Why a purely local rule is wrong}
        Flipping $C_1$ or $C_2$ changes links crossing ancestor splits.  Omitting
        $\mu_0$ or those ancestor factors generally breaks detailed balance.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Detailed Balance for Each Four-State Heat Bath}
  Fix $D$ and $u$, and let
  \[
    \mathcal O_u(\sigma)
      =\{\sigma^{00},\sigma^{10},\sigma^{01},\sigma^{11}\}.
  \]
  This orbit is the same when viewed from any $x\in\mathcal O_u(\sigma)$.
  The previous rule can be written
  \[
    K_u^D(x,y)
      =\ind{y\in\mathcal O_u(x)}
       \frac{\nu(y\mid D)}{\nu(\mathcal O_u(x)\mid D)}.
  \]

  \begin{block}{Detailed balance, line by line}
    For $x,y$ in the same orbit,
    \[
      \nu(x\mid D)K_u^D(x,y)
      =\frac{\nu(x\mid D)\nu(y\mid D)}
             {\nu(\mathcal O_u\mid D)}
      =\nu(y\mid D)K_u^D(y,x).
    \]
    If $x,y$ are in different orbits, both sides are zero.
  \end{block}

  Hence $K_u^D$ is reversible and preserves the \emph{same} conditional law
  $\nu(\,\cdot\mid D)$.  On the support of $\nu$, the orbit denominator is
  positive; candidates with zero conditional weight simply receive probability zero.
\end{frame}

\begin{frame}{Final Components Need Their Own Two-State Heat Bath}
  Internal nodes alone miss a singleton root.  For every final component
  $R\in\Pi_1$, also consider the two states $\sigma$ and $\sigma^R$, where all
  spins in $R$ are flipped.

  A root flip preserves every internal $\Lambda_u$, while the between-root term
  has already cancelled.  Therefore
  \[
    \frac{\nu(\sigma^R\mid D)}{\nu(\sigma\mid D)}
      =\frac{\mu_0(\sigma^R)}{\mu_0(\sigma)},
    \qquad
    \boxed{
    \mathbb P(\text{flip }R\mid\sigma,D)
      =\frac{\mu_0(\sigma^R)}
             {\mu_0(\sigma)+\mu_0(\sigma^R)}.}
  \]

  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Uniform prior}
        Each final component, including a singleton, is flipped with probability
        $1/2$: this is precisely the usual \textsc{Swendsen}--\textsc{Wang}
        recolouring at the cut $1$.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Why this matters}
        With two vertices in an unsatisfied state, $D$ has two singleton roots.
        Without root moves, the proposed chain would be the identity and could
        not converge from that state.
      \end{alertblock}
    \end{column}
  \end{columns}

  The internal four-state moves are the additional hierarchical moves.
\end{frame}

\begin{frame}{From Conditional Balance to Gibbs Balance}
  Let $\mathcal L(D)$ be any list of root moves and internal-node moves chosen
  from the full $D$ before the spins are updated.

  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Stationary sweep}
        Every deterministic composition of these heat baths preserves
        $\nu(\,\cdot\mid D)$:
        \[
          \nu(\,\cdot\mid D)K_D
          =\nu(\,\cdot\mid D).
        \]
        A deterministic sweep need not itself be reversible.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Reversible sweep}
        Choose the order of $\mathcal L(D)$ uniformly at random.  Every order
        and its reverse have the same probability, hence
        \[
          K_D^*=K_D
          \quad\text{in }L^2(\nu(\,\cdot\mid D)).
        \]
      \end{block}
    \end{column}
  \end{columns}

  For such a reversible conditional sweep,
  \[
    \begin{aligned}
      \mu(dx)P(x,dy)
      &=\int \mu(dx)\,\mathbb P(dD\mid x)\,K_D(x,dy)\\
      &=\int \nu(dD)\,\nu(dx\mid D)\,K_D(x,dy),
    \end{aligned}
  \]
  This is symmetric in $x,y$: the marginal hierarchical kernel satisfies
  detailed balance for $\mu$.

\end{frame}

\begin{frame}{Stopping the Updates and Truncating $D$ Are Different}
  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Full $D_{\leq1}$ is kept}
        Choose from $D$---before any flip---a fixed depth, threshold, subtree,
        or prefix of the traversal.  The resulting composition still preserves
        $\nu(\,\cdot\mid D)$, hence its marginal preserves $\mu$.

        \medskip
        An order or stopping rule that reacts to the spins produced during the
        sweep needs a separate invariance proof.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Only $D_{\leq t}$ is observed, $t<1$}
        This is a different auxiliary variable.  If
        $\Lambda_\infty^t(\sigma)$ is the satisfied weight between components
        of $\Pi_t$, then
        {\small
        \[
          \boxed{
          \begin{aligned}
          \nu(\sigma\mid D_{\leq t})\propto{}&\mu_0(\sigma)
          e^{(1-t)\Lambda_\infty^t(\sigma)}\\[-2pt]
          &\times\prod_{\beta_u\leq t}
          \Lambda_u(\sigma)e^{(1-\beta_u)\Lambda_u(\sigma)}.
          \end{aligned}}
        \]
        }
        The residual boundary factor disappears only at $t=1$.
      \end{alertblock}
    \end{column}
  \end{columns}

  \begin{center}
    \textbf{Exact conclusion:} a full sweep, or any $D$-measurable partial sweep,
    has Gibbs stationary; convergence requires a separate ergodicity argument.
  \end{center}
\end{frame}
