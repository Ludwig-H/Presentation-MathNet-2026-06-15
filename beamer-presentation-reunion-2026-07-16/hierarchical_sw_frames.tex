\begin{frame}{Exponential Clocks Turn \textsc{Swendsen}--\textsc{Wang} into a Hierarchy}
  \[
    \mu(\sigma)\propto
    \mu_0(\sigma)
    \exp\!\left[-\sum_{e\in E}|W_e|\ind{e\text{ is not satisfied}}\right].
  \]

  \begin{columns}[onlytextwidth,c]
    \begin{column}{0.46\textwidth}
      For each satisfied edge, independently,
      \[
        \boxed{\ \xi_e\sim\operatorname{Exp}(|W_e|)\ },
      \]
      and set $\xi_e=+\infty$ otherwise.

      At level $t$, keep the links with $\xi_e\leq t$.

      \begin{block}{Classical cut}
        \[
          \mathbb P(\xi_e\leq1)
          =1-e^{-|W_e|}.
        \]
      \end{block}
    \end{column}

\begin{frame}{The \textsc{Edwards}--\textsc{Sokal} Lift Gives the Exact Conditional Law}
  For an admissible $D_{\leq1}$,
  \[
    \boxed{
    \mathbb P(dD\mid\sigma)
      =e^{-\Lambda_\infty(\sigma)}
       \prod_{u}
       \Lambda_u(\sigma)e^{-\beta_u\Lambda_u(\sigma)}\,dD,}
  \]
  where $\Lambda_\infty$ is the satisfied weight between the final components.

  \begin{columns}[onlytextwidth,c]
    \begin{column}{0.42\textwidth}
      \begin{block}{Joint law}
        \[
          \nu(\sigma,D)
          \propto
          \mu_0(\sigma)e^{-U(\sigma)}
          \mathbb P(D\mid\sigma).
        \]
      \end{block}
      Its $\sigma$-marginal is the target Gibbs law $\mu$.
    \end{column}

    \begin{column}{0.12\textwidth}
      \centering
      \textcolor{inria-2024-rouge}{\Large $\Longrightarrow$}

      \scriptsize
      Bayes +\newline cancellation
    \end{column}

    \begin{column}{0.42\textwidth}
      \begin{block}{Conditional law}
        \[
          \boxed{
          \nu(\sigma\mid D)
          \propto
          \mu_0(\sigma)
          \prod_u
          \Lambda_u(\sigma)e^{(1-\beta_u)\Lambda_u(\sigma)}.}
        \]
      \end{block}
    \end{column}
  \end{columns}

  \begin{center}
    \textbf{This is the law used to flip clusters at every level.}
  \end{center}
\end{frame}

\begin{frame}{Each Local Move Satisfies Detailed Balance}
  For the four-state orbit
  \[
    \mathcal O_u(\sigma)
      =\{\sigma^{00},\sigma^{10},\sigma^{01},\sigma^{11}\},
  \]
  the previous rule is a heat bath.  Hence, for $x,y\in\mathcal O_u$,
  \[
    \boxed{
    \nu(x\mid D)K_u^D(x,y)
      =\frac{\nu(x\mid D)\nu(y\mid D)}
             {\nu(\mathcal O_u\mid D)}
      =\nu(y\mid D)K_u^D(y,x).}
  \]

  Every local heat bath preserves the same conditional law, so
  \[
    \nu(\,\cdot\mid D)K_D=\nu(\,\cdot\mid D)
    \qquad\Longrightarrow\qquad
    \boxed{\ \mu P=\mu.\ }
  \]

  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Fixed sweep}
        Stationary for Gibbs, but not necessarily reversible.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Symmetric random order}
        Every order and its reverse are equally likely: the whole dynamics
        satisfies detailed balance.
      \end{block}
    \end{column}
  \end{columns}

  \begin{center}
    \textbf{Main result: the hierarchical dynamics has Gibbs stationary.}
  \end{center}
\end{frame}

\begin{frame}{Stopping the Sweep: the Simple Rule}
  \begin{columns}[onlytextwidth,c]
    \begin{column}{0.47\textwidth}
      \begin{block}{\Large $\checkmark$ Keep the full $D_{\leq1}$}
        Sample the full dendrogram, then choose from $D$---before the flips---
        a depth, a subtree, or a prefix of the sweep.

        \vspace{0.5em}
        Every such partial sweep still preserves Gibbs:
        \[
          \boxed{\ \mu P_{\mathrm{partial}}=\mu.\ }
        \]
      \end{block}
    \end{column}

    \begin{column}{0.47\textwidth}
      \begin{alertblock}{\Large $\boldsymbol{\neq}$ Observe only $D_{\leq t}$}
        Actually truncating the auxiliary variable at $t<1$ gives a different
        conditional law, with the residual factor
        \[
          \boxed{\ e^{(1-t)\Lambda_\infty^t(\sigma)}.\ }
        \]

        The full-$D$ flip probabilities cannot simply be reused.
      \end{alertblock}
    \end{column}
  \end{columns}

  \begin{center}
    \textbf{Stop the updates if desired; do not silently change the auxiliary law.}
  \end{center}
\end{frame}
