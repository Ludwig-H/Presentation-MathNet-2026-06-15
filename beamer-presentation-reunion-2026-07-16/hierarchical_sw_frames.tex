\begin{frame}{Exponential Clocks Replace Rigid Freezing}
  \begin{columns}[onlytextwidth,c]
    \begin{column}{0.43\textwidth}
      An edge $e=\{i,j\}$ is \textbf{satisfied} when
      \[
        \begin{array}{ll}
          \sigma_i=\sigma_j, & e\in F,\\[2pt]
          \sigma_i\neq\sigma_j, & e\in\widetilde F.
        \end{array}
      \]

      For each satisfied edge, independently,
      \[
        \boxed{\ \xi_e\sim\operatorname{Exp}(|W_e|)\ }.
      \]
      At level $t$, keep the edges such that $\xi_e\leq t$;
      unsatisfied edges are never kept.

      \begin{block}{Classical \textsc{Swendsen}--\textsc{Wang}}
        The usual freezing rule is exactly the cut $t=1$:
        \[
          \mathbb P(\xi_e\leq1)=1-e^{-|W_e|}.
        \]
      \end{block}
    \end{column}

\begin{frame}{Keep the Merge, Forget the Winning Edge}
  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.45\textwidth}
      \begin{block}{What the dendrogram $D$ keeps}
        \begin{itemize}
          \item the nested partitions of the vertices;
          \item the order and levels of the merges;
          \item the cut $t=1$ separating the final forest components.
        \end{itemize}
      \end{block}

      \begin{alertblock}{What $D$ does not keep}
        The identity of the edge selected by Kruskal at a merge.
        Keeping it would hard-constrain the two children and reduce the update
        to ordinary component recolouring.
      \end{alertblock}
    \end{column}

    \begin{column}{0.51\textwidth}
      At a merge $C=C_1\mathbin{\dot\cup}C_2$, the exponential clocks form a race.
      After summing over the winning edge, its rate is simply
      \[
        \boxed{
        \operatorname{rate}(C_1,C_2\mid\sigma)
        =\!\!\sum_{\substack{e=\{i,j\},\ i\in C_1,\ j\in C_2\\
                         \sigma\ \mathrm{satisfies}\ e}}
          |W_e|.}
      \]
      Multiplying these exponential-race densities along $D$ defines
      \[
        \mathbb P(\mathrm dD\mid\sigma)
      \]
      while marginalising the irrelevant winning edge.

      The same $D$ may therefore have positive probability for $\sigma$ and
      after a subtree flip. If not, the conditional heat bath assigns that
      flip probability zero.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{An \textsc{Edwards}--\textsc{Sokal} Law on $(\sigma,D)$}
  Let $\mathbb P(\mathrm dD\mid\sigma)$ be exactly the clock--Kruskal law from
  the previous slides.  On pairs $(\sigma,D)$, define
  \[
    \boxed{
      \mu(\mathrm d\sigma,\mathrm dD)
      =\frac{1}{Z}\,
        \mu_0(\mathrm d\sigma)\,e^{-U(\sigma)}
        \mathbb P(\mathrm dD\mid\sigma)
      =\mu(\mathrm d\sigma)\mathbb P(\mathrm dD\mid\sigma).}
  \]

  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Correct marginal}
        Since $\mathbb P(\,\cdot\mid\sigma)$ is a probability law,
        \[
          \int \mu(\mathrm d\sigma,\mathrm dD)
          =\frac{1}{Z}\mu_0(\mathrm d\sigma)e^{-U(\sigma)}
          =\mu(\mathrm d\sigma).
        \]
      \end{block}
    \end{column}

    \begin{column}{0.49\textwidth}
      \begin{block}{Conditional law used for recolouring}
        For a fixed dendrogram,
        \[
          \mu(\mathrm d\sigma\mid D)
          \propto
          \mu_0(\mathrm d\sigma)e^{-U(\sigma)}
          \mathbb P(\mathrm dD\mid\sigma).
        \]
        The prior $\mu_0$ is therefore preserved exactly; it need not be uniform.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

    \begin{column}{0.52\textwidth}
      At the current split $C=C_1\mathbin{\dot\cup}C_2$, compare only two
      configurations:
      \begin{itemize}
        \item the current configuration $\sigma$;
        \item $\sigma^{C_1}$, obtained by reversing every label in $C_1$.
      \end{itemize}
      Replace $\sigma$ by $\sigma^{C_1}$ with the conditional heat-bath probability
      \[
        \boxed{
        \frac{\mu(\sigma^{C_1}\mid D)}
             {\mu(\sigma\mid D)+\mu(\sigma^{C_1}\mid D)}.}
      \]
      Then update $\sigma$, descend to the next split, and recompute the
      probability.

      \begin{alertblock}{The essential point}
        The choices at different levels are sequential conditional updates,
        not independent flips.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Gibbs Measure Is Stationary by Construction}
  \begin{block}{One split satisfies detailed balance}
    For the two configurations $\sigma$ and $\sigma^C$,
    \[
      \mu(\sigma\mid D)\,
      \mathbb P(\sigma\to\sigma^C\mid D)
      =\frac{\mu(\sigma\mid D)\mu(\sigma^C\mid D)}
             {\mu(\sigma\mid D)+\mu(\sigma^C\mid D)}
      =\mu(\sigma^C\mid D)\,
       \mathbb P(\sigma^C\to\sigma\mid D).
    \]
    Hence every split preserves $\mu(\,\cdot\mid D)$, and so does the complete
    top-down sweep.
  \end{block}

  \begin{center}
    \Large
    $\displaystyle
      \sigma
      \ \xrightarrow{\ D\sim\mathbb P(\,\cdot\mid\sigma)\ }\
      (\sigma,D)
      \ \xrightarrow{\ \text{top-down heat baths}\ }\
      (\sigma',D)
      \ \xrightarrow{\ \text{discard }D\ }\
      \sigma'.$
  \end{center}

  \begin{block}{Whole transition}
    The first arrow samples one conditional law of
    $\mu(\mathrm d\sigma,\mathrm dD)$; the sweep preserves the other conditional law.
    This is a Gibbs-within-Gibbs update on $(\sigma,D)$. Therefore
    \[
      \boxed{\ \sigma\sim\mu\quad\Longrightarrow\quad\sigma'\sim\mu.\ }
    \]
  \end{block}

\end{frame}
