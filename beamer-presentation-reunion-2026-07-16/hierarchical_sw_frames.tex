\begin{frame}{Why the Hierarchical Dynamics Preserves Gibbs}
  \small
  For the one-cluster kernel $H_A^D$,
  \[
    \nu_D(\sigma)H_A^D(\sigma,F_A\sigma)
    =\frac{\nu_D(\sigma)\nu_D(F_A\sigma)}
           {\nu_D(\sigma)+\nu_D(F_A\sigma)}
    =\nu_D(F_A\sigma)H_A^D(F_A\sigma,\sigma).
  \]
  Thus every $H_A^D$ is $\nu_D$-reversible. Any full sweep $Q_D$ is
  $\nu_D$-stationary; a random scan or a palindromic sweep is also reversible.

  After refreshing $D$, the spin kernel is
  \[
    K(\sigma,\sigma')=\int\nu(dD\mid\sigma)Q_D(\sigma,\sigma').
  \]
  Therefore
  \[
    \sum_\sigma\mu(\sigma)K(\sigma,\sigma')
    =\int\nu(dD)\sum_\sigma\nu_D(\sigma)Q_D(\sigma,\sigma')
    =\int\nu(dD)\nu_D(\sigma')=\mu(\sigma').
  \]
  \begin{block}{Conclusion}
    The spin chain is Markov and Gibbs-stationary. If $Q_D$ is reversible,
    the same computation with $(\sigma,\sigma')$ proves detailed balance for $K$.
  \end{block}
\end{frame}

\begin{frame}{A Canonical Lift: Censored Edwards--Sokal Clocks}
  \small
  Set $J_e=|W_e|$, $\eta_e=\operatorname{sgn}(W_e)$ and
  \[
    U(\sigma)=\sum_{e=\{i,j\}}J_e\ind{\sigma_i\sigma_j\neq\eta_e},
    \qquad \mu(\sigma)=Z^{-1}e^{-U(\sigma)}.
  \]
  For independent $X_e\sim\mathrm{Exp}(J_e)$, retain
  $T_e=X_e$ only when $\sigma_i\sigma_j=\eta_e$ and $X_e\leq1$;
  otherwise set $T_e=\infty$. Equivalently,
  \[
    \overline\mu(\sigma,dT)=\frac1Z\prod_{e\in E}\left[
      e^{-J_e}\delta_\infty(dT_e)
      +\ind{\sigma_i\sigma_j=\eta_e}
       J_e e^{-J_eT_e}\ind{0\leq T_e\leq1}\,dT_e
    \right].
  \]
  Each factor integrates to
  \[
    \begin{cases}
      e^{-J_e}+\int_0^1J_e e^{-J_et}\,dt=1,
        &\sigma_i\sigma_j=\eta_e,\\
      e^{-J_e},&\sigma_i\sigma_j\neq\eta_e,
    \end{cases}
  \]
  so $\overline\mu(\sigma)=\mu(\sigma)$. Moreover, with
  $A(T)=\{e:T_e<\infty\}$,
  \[
    \overline\mu(\sigma\mid T)=2^{-c(A)}\ind{\sigma\in\Omega_A},
    \qquad
    \Omega_A=\{\tau:\tau_i\tau_j=\eta_e\ \forall e\in A\}.
  \]
\end{frame}

\begin{frame}{Consequences for Weak Recovery}
  \begin{columns}[onlytextwidth,T]
    \begin{column}{0.48\textwidth}
      \begin{block}{What is rigorous}
        The heat-bath dendrogram sweep is Gibbs-stationary.
        For the censored edge lift, however, its exact MSF specialization has
        exactly the SW component law.
        On the homogeneous triangular GSBM, in the truth coupling,
        \[
          \mathbb P(e\text{ active})=2p-1,
        \]
        \[
          p_c^{\mathrm{ES\text{-}MSF}}
          =p_c^{\mathrm{SW}}
          =\frac12+\sin\!\left(\frac{\pi}{18}\right)
          \simeq0.67365.
        \]
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{What a strict improvement requires}
        To beat the triangle certificate
        \[
          p_c^\triangle=\frac{7-\sqrt{17}}4\simeq0.71922,
        \]
        one needs a genuinely new positive macro-block lift $\nu(\sigma,D)$
        (so proper subtree flips have non-zero conditional weight), together
        with a new boundary-connectivity/percolation proof.

        \centering\textbf{Hierarchy alone is not a new threshold.}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
